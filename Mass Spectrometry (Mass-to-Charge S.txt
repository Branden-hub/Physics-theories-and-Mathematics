import json, re, uuid
from hashlib import sha1

DOMAINS = {
  "nlp": ["token", "language", "bert", "gpt", "llm", "text", "qa", "ner", "summarization"],
  "cv": ["image", "vision", "opencv", "segmentation", "detection", "pose"],
  "audio": ["speech", "asr", "tts", "audio", "whisper"],
  "multimodal": ["clip", "vision-language", "vlm", "mm"],
  "reasoning": ["agent", "tool", "planning", "chain-of-thought", "symbolic"],
  "robotics": ["ros", "manipulation", "kinematics"],
  "data": ["etl", "dataset", "preprocessing"],
  "infra": ["inference", "deployment", "quantization", "optimization", "serving"]
}

def stable_id(source: str, name: str, repo_url: str) -> str:
    basis = f"{source}|{name}|{repo_url}"
    return str(uuid.uuid5(uuid.NAMESPACE_URL, sha1(basis.encode()).hexdigest()))

def infer_domains(text: str):
    text_low = text.lower()
    hits = []
    for dom, keys in DOMAINS.items():
        if any(k in text_low for k in keys):
            hits.append(dom.upper())
    return sorted(set(hits)) or ["N/A"]

def normalize_component(raw: dict) -> dict:
    text = " ".join(filter(None, [
        raw.get("name",""), raw.get("description",""), raw.get("readme","")
    ]))
    domains = infer_domains(text)
    return {
      "id": stable_id(raw["source"], raw.get("name",""), raw.get("repo_url","")),
      "name": raw.get("name"),
      "summary": (raw.get("description") or "")[:220],
      "domain": domains,
      "skills": raw.get("skills", []),
      "source": raw["source"],
      "repo_url": raw.get("repo_url"),
      "homepage": raw.get("homepage"),
      "docs_url": raw.get("docs_url"),
      "license": raw.get("license"),
      "languages": raw.get("languages", []),
      "tags": raw.get("tags", []),
      "dependencies": raw.get("dependencies", []),
      "example_code": (raw.get("example_code") or "")[:2000],
      "readme_excerpt": (raw.get("readme") or "")[:1500],
      "stars": raw.get("stars", 0),
      "last_commit_iso": raw.get("last_commit_iso"),
      "is_active": 1 if raw.get("is_active", True) else 0,
      "added_at_iso": raw.get("added_at_iso"),
      "updated_at_iso": raw.get("updated_at_iso"),
      "deprecated": 1 if raw.get("deprecated") else 0,
      "broken_links": raw.get("broken_links", [])
    }


# storage.py
import json, sqlite3, time
from contextlib import contextmanager

CREATE_TABLE = """..."""  # from previous section

@contextmanager
def db(path="components.db"):
    con = sqlite3.connect(path)
    try:
        yield con
    finally:
        con.commit()
        con.close()

def init_db():
    with db() as con:
        con.executescript(CREATE_TABLE)

def upsert_component(con, comp: dict):
    keys = list(comp.keys())
    cols = ",".join(keys)
    placeholders = ",".join(["?"]*len(keys))
    updates = ",".join([f"{k}=excluded.{k}" for k in keys if k != "id"])
    vals = [json.dumps(v) if isinstance(v, (list, dict)) else v for v in comp.values()]
    sql = f"INSERT INTO components ({cols}) VALUES ({placeholders}) ON CONFLICT(id) DO UPDATE SET {updates}"
    con.execute(sql, vals)


# utils.py
import os, time, random, functools, requests
from dotenv import load_dotenv

load_dotenv()
UA = os.getenv("USER_AGENT", "AI-Component-Aggregator/1.0")

def jitter(base=1.0):
    return base + random.random()

def retry(max_retries=3, backoff=2.0):
    def deco(fn):
        @functools.wraps(fn)
        def wrap(*a, **kw):
            for i in range(max_retries):
                try:
                    return fn(*a, **kw)
                except Exception:
                    if i == max_retries - 1:
                        raise
                    time.sleep((backoff ** i) * jitter())
        return wrap
    return deco

@retry()
def get(url, headers=None, params=None):
    h = {"User-Agent": UA}
    if headers: h.update(headers)
    r = requests.get(url, headers=h, params=params, timeout=30)
    r.raise_for_status()
    return r


# fetchers/github.py
import os, datetime
from utils import get
from normalize import normalize_component

GITHUB_TOKEN = os.getenv("GITHUB_TOKEN")

def gh_headers():
    h = {"Accept": "application/vnd.github+json"}
    if GITHUB_TOKEN:
        h["Authorization"] = f"Bearer {GITHUB_TOKEN}"
    return h

def search_repos(query, per_page=50):
    url = "https://api.github.com/search/repositories"
    data = get(url, headers=gh_headers(), params={"q": query, "per_page": per_page, "sort":"stars","order":"desc"}).json()
    return data.get("items", [])

def get_readme(owner, repo):
    url = f"https://api.github.com/repos/{owner}/{repo}/readme"
    r = get(url, headers=gh_headers())
    if r.ok:
        j = r.json()
        if j.get("download_url"):
            return get(j["download_url"]).text
    return ""

def hydrate_repo(repo):
    owner, name = repo["full_name"].split("/")
    readme = get_readme(owner, name)
    return {
      "source": "github",
      "name": repo["name"],
      "description": repo.get("description"),
      "repo_url": repo["html_url"],
      "homepage": repo.get("homepage"),
      "license": (repo.get("license") or {}).get("spdx_id"),
      "languages": [],  # optional: GET /repos/{owner}/{repo}/languages
      "tags": [],
      "dependencies": [],
      "example_code": "",
      "readme": readme,
      "stars": repo.get("stargazers_count", 0),
      "last_commit_iso": repo.get("pushed_at"),
      "is_active": True,
      "added_at_iso": datetime.datetime.utcnow().isoformat(),
      "updated_at_iso": datetime.datetime.utcnow().isoformat(),
      "deprecated": repo.get("archived") or repo.get("disabled") or False,
      "broken_links": []
    }

def fetch(config):
    comps = []
    for q in config["sources"]["github"]["queries"]:
        items = search_repos(q, per_page=config["sources"]["github"]["per_query_limit"])
        for repo in items:
            raw = hydrate_repo(repo)
            comps.append(normalize_component(raw))
    return comps

# exporters/export_jsonl.py
import json

def export_jsonl(rows, path="export/components.jsonl"):
    import os
    os.makedirs("export", exist_ok=True)
    with open(path, "w", encoding="utf-8") as f:
        for r in rows:
            f.write(json.dumps(r, ensure_ascii=False) + "\n")

# exporters/export_markdown.py
from collections import defaultdict

def export_markdown(rows, path="export/index.md"):
    import os
    os.makedirs("export", exist_ok=True)
    rows_by_domain = defaultdict(list)
    for r in rows:
        for d in r["domain"]:
            rows_by_domain[d].append(r)
    with open(path, "w", encoding="utf-8") as f:
        f.write("# AI Component Library\n\n")
        for dom in sorted(rows_by_domain.keys()):
            f.write(f"## {dom}\n\n")
            for r in rows_by_domain[dom]:
                name = r['name'] or 'Unnamed'
                repo = r.get('repo_url') or r.get('homepage') or ''
                summary = r.get('summary') or ''
                f.write(f"- **{name}** — {summary} [{repo}]\n")
            f.write("\n")


# main.py
import os, yaml
from storage import init_db, db, upsert_component
from fetchers import github, huggingface, langchain_hub, haystack, awesome_lists
from exporters.export_jsonl import export_jsonl
from exporters.export_markdown import export_markdown

FETCHERS = [github, huggingface, langchain_hub, haystack, awesome_lists]

def load_config(path="config.yml"):
    with open(path, "r", encoding="utf-8") as f:
        return yaml.safe_load(f)

def run():
    config = load_config()
    init_db()
    all_rows = []
    for mod in FETCHERS:
        try:
            rows = mod.fetch(config)
            all_rows.extend(rows)
        except Exception as e:
            print(f"[WARN] fetcher {mod.__name__} failed: {e}")

    # dedupe by id in-memory before DB upsert
    seen = {}
    for r in all_rows:
        seen[r["id"]] = r
    all_rows = list(seen.values())

    with db() as con:
        for r in all_rows:
            upsert_component(con, r)

    export_jsonl(all_rows)
    export_markdown(all_rows)
    print(f"Done. Collected {len(all_rows)} components.")

if __name__ == "__main__":
    run()
