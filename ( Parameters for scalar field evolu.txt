(* Parameters for scalar field evolution *)
params = <|
   "diffusion" -> 0.2,
   "alpha" -> 0.5,
   "beta" -> Pi,
   "gamma" -> 2,
   "lambda" -> 0.5,
   "zeta" -> 0.1,
   "maxSteps" -> 100,
   "convergenceThreshold" -> 10^-6
|>;

(* Sample 3-SAT problem: (x1 ∨ ¬x2 ∨ x3) ∧ (¬x1 ∨ x2 ∨ x4) ∧ (x2 ∨ ¬x3 ∨ ¬x4) *)
clauses = {
   {1, -2, 3},
   {-1, 2, 4},
   {2, -3, -4}
};

(* Initial random field values in [0, Pi] *)
seed = 42;
SeedRandom[seed];
vars = Range[4];
phiInit = AssociationThread[vars, RandomReal[{0, Pi}, Length[vars]]];

(* Clause energy function based on cosine penalty *)
clauseEnergy[phi_, clause_] := Times @@ (1 - Cos[phi[Abs[#]] * Sign[#]] & /@ clause)
totalEnergy[phi_] := Total[clauseEnergy[phi, #] & /@ clauses]

(* Evolution step *)
evolve[phi_, step_] := Module[{newPhi = phi, grad, v},
  Do[
    v = phi[x];
    grad = -params["alpha"]*v*(v^2 - params["beta"]^2);
    grad += -params["gamma"]*D[totalEnergy[phi /. {x -> v}], v];
    newPhi[x] = v + params["diffusion"]*grad;
    ,
    {x, vars}
  ];
  newPhi
];

(* Main evolution loop *)
phiStates = {phiInit};
converged = False;

Do[
  newPhi = evolve[Last[phiStates], step];
  phiStates = Append[phiStates, newPhi];
  delta = Max[Abs[Values[newPhi] - Values[Last[phiStates]]]];
  If[delta < params["convergenceThreshold"],
   converged = True;
   Break[];
  ],
  {step, params["maxSteps"]}
];

(* Decode final phi values to Boolean assignments *)
finalPhi = Last[phiStates];
assignments = AssociationMap[If[finalPhi[#] > Pi/2, 1, 0] &, vars];

(* Check clause satisfaction *)
satisfiedQ[clause_, assignment_] := Or @@ (If[Sign[l] > 0, assignment[Abs[l]] == 1, assignment[Abs[l]] == 0] & /@ clause)
allSatisfied = And @@ (satisfiedQ[#, assignments] & /@ clauses);

(* Report *)
Grid[{
  {"Final φ values", finalPhi},
  {"Boolean assignment", assignments},
  {"All clauses satisfied?", allSatisfied},
  {"Converged in steps", Length[phiStates] - 1}
}]
